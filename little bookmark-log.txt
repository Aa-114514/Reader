javascript:(function(){
if(document.getElementById('RDR_LOGVIEW')){document.getElementById('RDR_LOGVIEW').parentNode.removeChild(document.getElementById('RDR_LOGVIEW'));return}
var raw=localStorage.getItem('RDR_TAPLOG');
if(!raw){alert('无日志');return}
var arr;try{arr=JSON.parse(raw)}catch(e){alert('日志解析失败');return}
if(!arr||!arr.length){alert('日志为空');return}
var O=document.createElement('div');O.id='RDR_LOGVIEW';
O.style.cssText='position:fixed;top:0;left:0;right:0;bottom:0;background:rgba(0,0,0,.96);color:#0f0;font:12px/1.5 monospace;z-index:2147483647;overflow:auto;-webkit-overflow-scrolling:touch;padding:0;box-sizing:border-box';
function esc(s){return String(s).replace(/[&<>"]/g,function(c){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]})}
function pad(n){return ('0'+n).slice(-2)}
function buildText(){
  var t='RDR 点击日志 ('+arr.length+')\n';
  t+='导出时间: '+new Date().toLocaleString()+'\n';
  t+='页面: '+location.href+'\n';
  t+='UA: '+navigator.userAgent+'\n';
  t+='-------\n';
  arr.slice().reverse().forEach(function(r,i){
    var d=new Date(r.t);
    var ts=pad(d.getHours())+':'+pad(d.getMinutes())+':'+pad(d.getSeconds());
    t+=(i+1)+' ['+ts+']['+r.e+'] '+r.tg+(r.id?'#'+r.id:'')+(r.cl?'.'+r.cl.split(/\s+/).join('.'):'')+'\n';
    if(r.tx)t+='    文本: '+r.tx+'\n';
    t+='    路径: '+r.pt+'\n';
  });
  return t;
}
function copyAll(){
  var txt=buildText();
  try{
    navigator.clipboard.writeText(txt).then(function(){
      var btn=O.querySelector('#rllCopy');
      btn.textContent='✓ 已复制';
      setTimeout(function(){btn.textContent='📋 复制全部'},1500);
    }).catch(function(){fallbackCopy(txt)});
  }catch(e){fallbackCopy(txt)}
}
function fallbackCopy(txt){
  var ta=document.createElement('textarea');
  ta.value=txt;
  ta.style.cssText='position:fixed;left:-9999px;top:0';
  document.body.appendChild(ta);
  ta.select();
  try{document.execCommand('copy');
    var btn=O.querySelector('#rllCopy');
    btn.textContent='✓ 已复制';
    setTimeout(function(){btn.textContent='📋 复制全部'},1500);
  }catch(e){alert('复制失败，请长按选择')}
  document.body.removeChild(ta);
}
var h='<div style="position:sticky;top:0;background:rgba(0,0,0,.98);padding:8px 10px;display:flex;gap:6px;border-bottom:1px solid #333;z-index:2;flex-wrap:wrap"><span style="flex:1;color:#0f0;min-width:120px">RDR 点击日志 ('+arr.length+')</span><button id="rllCopy" style="background:#2ecc71;color:#fff;border:0;border-radius:6px;padding:6px 10px;font-size:12px">📋 复制全部</button><button id="rllClear" style="background:#4a8fe0;color:#fff;border:0;border-radius:6px;padding:6px 10px;font-size:12px">清空</button><button id="rllClose" style="background:#444;color:#fff;border:0;border-radius:6px;padding:6px 10px;font-size:12px">关闭</button></div><div style="padding:8px 10px">';
arr.slice().reverse().forEach(function(r,i){
  var t=new Date(r.t);
  var ts=pad(t.getHours())+':'+pad(t.getMinutes())+':'+pad(t.getSeconds());
  h+='<div style="border-bottom:1px dashed #333;padding:6px 0">';
  h+='<div style="color:#0ff">'+(i+1)+' ['+ts+']['+esc(r.e)+'] '+esc(r.tg)+(r.id?'#'+esc(r.id):'')+(r.cl?'.'+esc(r.cl.split(/\s+/).join('.')):'')+'</div>';
  if(r.tx)h+='<div style="color:#ff0;padding-left:8px">'+esc(r.tx)+'</div>';
  h+='<div style="color:#888;padding-left:8px;word-break:break-all">'+esc(r.pt)+'</div>';
  h+='</div>';
});
h+='</div>';
O.innerHTML=h;
(document.body||document.documentElement).appendChild(O);
O.querySelector('#rllClose').onclick=function(){O.parentNode.removeChild(O)};
O.querySelector('#rllClear').onclick=function(){if(confirm('清空日志？')){localStorage.removeItem('RDR_TAPLOG');O.parentNode.removeChild(O)}};
O.querySelector('#rllCopy').onclick=copyAll;
})();
